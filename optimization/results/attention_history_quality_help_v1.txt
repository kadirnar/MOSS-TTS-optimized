  --compressed-scales   Lossless SM90 compression of selected FP32 scale
                        allocations
  --short-scales        Exact BF16 scale storage with explicit Gluon reduction
                        layouts
  --projection-pdl      SM90 projection dependency overlap with exact short
                        scales
  --attention-pdl       Extend SM90 dependency overlap through native
                        attention and quantization
  --codec-clock         Exact FP32 codec with shared stage counters and first-
                        frame specialization
  --first-audio-graph   Run the initial 32 audio steps in one CUDA graph
  --bulk-prefetch       Prefix-only asynchronous bulk L2 hints for selected
                        projections
  --prefill-qkv         Fuse BF16 prefill Q/K normalization, rotary and KV
                        writes
  --prefill-pointwise   Fuse BF16 prefill activation/product and
                        residual/normalization
  --async-output        Stage exact G32 output weights with SM90 asynchronous
                        copies
  --output-weight-prefetch
                        Preload exact G32 attention-output weights into
                        registers before PDL wait
  --qkv-cluster         Fuse G32 QKV/head preparation using SM90 clustered
                        cubins
  --down-tile8          Use eight-row/four-warp G32 down projections with
                        clustered QKV
  --gateup-compiler     Use exact one-CTA gate/up cubins from isolated Triton
                        3.8
  --attention-history   Preload historical KV entries before the attention
                        dependency wait
  --g64-norm-projections {up,up_qkv}
                        Experimental G64 weights/G32 activations; changes
                        audio and requires quality review
