options:
  -h, --help            show this help message and exit
  --port PORT
  --attention-block {32,128}
  --fused-residual
  --fused-gateup
  --weight-quantization {none,fp8_all}
                        FP8 is experimental; all 32 codebooks and FP32 codec
                        are retained
  --calibration CALIBRATION
                        Experimental complete GPTQ export: grouped DP4A,
                        audited normalization layout, fused gate/up, 160-token
                        prefill bucket
  --packing-plan PACKING_PLAN
                        Optional validated INT4 packing plan for a calibrated
                        export
  --decode-buckets      Capture position-bounded decode attention graphs
  --metrics-file METRICS_FILE
                        Optional local JSONL stage timing diagnostics;
                        excludes text/audio/voice IDs
  --attention-quant     Fuse G32 activation quantization into attention
                        reduction
  --native-attention    Use SM90 native B32/W4 attention with fixed arithmetic
                        order
  --gateup-quant        Fuse grouped activation quantization into gate/up/SiLU
  --scaled-dp4a         Use exact scaled-integer INT4 unpacking
  --norm-projection     Fuse residual normalization and quantization into QKV
                        and gate/up projections
  --audio-head-buckets  Skip only currently masked future audio heads; retain
                        all 32 codebooks per frame
  --compressed-scales   Lossless SM90 compression of selected FP32 scale
                        allocations
  --short-scales        Exact BF16 scale storage with explicit Gluon reduction
                        layouts
  --projection-pdl      SM90 projection dependency overlap with exact short
                        scales
  --attention-pdl       Extend SM90 dependency overlap through native
                        attention and quantization
  --codec-clock         Exact FP32 codec with shared stage counters and first-
                        frame specialization
  --first-audio-graph   Run the initial 32 audio steps in one CUDA graph
  --bulk-prefetch       Prefix-only asynchronous bulk L2 hints for selected
                        projections
  --prefill-qkv         Fuse BF16 prefill Q/K normalization, rotary and KV
                        writes
  --prefill-pointwise   Fuse BF16 prefill activation/product and
                        residual/normalization
  --async-output        Stage exact G32 output weights with SM90 asynchronous
                        copies
  --output-weight-prefetch
                        Preload exact G32 attention-output weights into
                        registers before PDL wait
  --qkv-cluster         Fuse G32 QKV/head preparation using SM90 clustered
                        cubins
  --down-tile8          Use eight-row/four-warp G32 down projections with
                        clustered QKV
  --gateup-compiler     Use exact one-CTA gate/up cubins from isolated Triton
                        3.8
  --attention-history   Preload historical KV entries before the attention
                        dependency wait
  --cuda-only-seed      Seed only the current CUDA device used by the
                        inference worker
